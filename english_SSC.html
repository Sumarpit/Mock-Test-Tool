let dataset = []; let currentMode = 'vocab'; let filteredList = []; let currentIndex = 0; let score = 0; let streak = 0; // Fetch data automatically on load async function init() { try { const response = await fetch('./data.json'); if (!response.ok) throw new Error('Network response was not ok'); dataset = await response.json(); // Hide the manual load section if fetch is successful document.querySelector('.data-section').style.display = 'none'; filterData(); renderCard(); } catch (error) { console.error('Failed to load data.json:', error); document.getElementById('target-word').innerText = "Fetch Error"; document.getElementById('sub-info').innerText = "Could not load data.json. Ensure it is in the same directory."; } } function switchMode(mode) { currentMode = mode; document.querySelectorAll('.tab-btn').forEach(btn => btn.classList.remove('active')); event.target.classList.add('active'); if (mode === 'vocab') { document.getElementById('vocab-ui').classList.remove('hidden'); document.getElementById('spelling-ui').classList.add('hidden'); } else { document.getElementById('vocab-ui').classList.add('hidden'); document.getElementById('spelling-ui').classList.remove('hidden'); } currentIndex = 0; filterData(); renderCard(); } // Routes objects based on their JSON keys function filterData() { if (currentMode === 'vocab') { filteredList = dataset.filter(item => item.o !== undefined); // Has options array } else { filteredList = dataset.filter(item => item.f !== undefined); // Has boolean flag } } function renderCard() { if (filteredList.length === 0) { document.getElementById('target-word').innerText = "Complete"; document.getElementById('sub-info').innerText = "No more words in this category."; document.getElementById('options-container').innerHTML = ""; return; } const item = filteredList[currentIndex]; document.getElementById('target-word').innerText = item.w; if (currentMode === 'vocab') { document.getElementById('sub-info').innerText = "Select the correct synonym/meaning:"; const container = document.getElementById('options-container'); container.innerHTML = ""; item.o.forEach((opt, idx) => { const btn = document.createElement('button'); btn.className = 'option-btn'; btn.innerHTML = `${opt}${idx + 1}`; btn.onclick = () => handleVocabChoice(idx); container.appendChild(btn); }); } else { document.getElementById('sub-info').innerText = item.m ? `Hint: ${item.m}` : "Is this spelling correct?"; } } function handleVocabChoice(selectedIndex) { const item = filteredList[currentIndex]; const isCorrect = selectedIndex === item.a; processResult(isCorrect); } function handleBinaryChoice(userChoice) { const item = filteredList[currentIndex]; const isCorrect = userChoice === item.f; if (!isCorrect && item.c && !item.f) { alert(`Incorrect! Correct spelling is: ${item.c}`); } processResult(isCorrect); } function processResult(isCorrect) { const container = document.getElementById('app-container'); if (isCorrect) { score += 10; streak += 1; container.classList.add('flash-correct'); } else { streak = 0; container.classList.add('flash-wrong'); } document.getElementById('score').innerText = score; document.getElementById('streak').innerText = streak; setTimeout(() => { container.classList.remove('flash-correct', 'flash-wrong'); currentIndex = (currentIndex + 1) % filteredList.length; renderCard(); }, 200); } // Keyboard controls window.addEventListener('keydown', (e) => { if (currentMode === 'vocab') { if (['1', '2', '3', '4'].includes(e.key)) { handleVocabChoice(parseInt(e.key) - 1); } } else if (currentMode === 'spelling') { if (e.key === 'ArrowLeft') handleBinaryChoice(false); if (e.key === 'ArrowRight') handleBinaryChoice(true); } }); // Start app init();
